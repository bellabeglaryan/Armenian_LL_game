-- 1. Basic patient directory with user contact information SELECT p.patient_id, u.name, u.surname, u.email, u.phone_number, p.gender, EXTRACT(YEAR FROM age(CURRENT_DATE, p.birth_date)) AS age, p.diagnosis, p.registration_date FROM patients p JOIN users u ON u.user_id = p.patient_id ORDER BY p.registration_date DESC, p.patient_id; -- 2. Active therapist directory SELECT t.therapist_id, u.name, u.surname, u.email, t.specialization, t.experience_years, t.license_number FROM therapists t JOIN users u ON u.user_id = t.therapist_id WHERE t.is_active = TRUE ORDER BY t.specialization, t.experience_years DESC; -- 3. Patients and their assigned therapists SELECT p.patient_id, pu.name || ' ' || pu.surname AS patient_name, p.diagnosis, t.therapist_id, tu.name || ' ' || tu.surname AS therapist_name, t.specialization FROM patient_therapist_map ptm JOIN patients p ON p.patient_id = ptm.patient_id JOIN users pu ON pu.user_id = p.patient_id JOIN therapists t ON t.therapist_id = ptm.therapist_id JOIN users tu ON tu.user_id = t.therapist_id ORDER BY patient_name; -- 4. Patient emergency contacts SELECT p.patient_id, pu.name || ' ' || pu.surname AS patient_name, ec.name || ' ' || ec.surname AS contact_name, ec.relation, ec.phone_number, ec.email FROM emergency_contacts ec JOIN patients p ON p.patient_id = ec.patient_id JOIN users pu ON pu.user_id = p.patient_id ORDER BY patient_name, ec.relation; -- 5. Appointment history with patient, therapist, and slot details SELECT a.appointment_id, a.scheduled_at, a.status, pu.name || ' ' || pu.surname AS patient_name, tu.name || ' ' || tu.surname AS therapist_name, ts.start_time, ts.end_time FROM appointments a JOIN users pu ON pu.user_id = a.patient_id JOIN users tu ON tu.user_id = a.therapist_id LEFT JOIN time_slots ts ON ts.slot_id = a.slot_id ORDER BY a.scheduled_at DESC; -- 6. Appointment counts by status SELECT status, COUNT(*) AS appointment_count, ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS percentage_of_total FROM appointments GROUP BY status ORDER BY appointment_count DESC; -- 7. Monthly appointment volume SELECT date_trunc('month', scheduled_at)::date AS appointment_month, COUNT(*) AS total_appointments, COUNT(*) FILTER (WHERE status = 'completed') AS completed_appointments, COUNT(*) FILTER (WHERE status = 'cancelled') AS cancelled_appointments, COUNT(*) FILTER (WHERE status = 'no_show') AS no_show_appointments FROM appointments GROUP BY date_trunc('month', scheduled_at) ORDER BY appointment_month; -- 8. Therapist workload and completion rate SELECT t.therapist_id, u.name || ' ' || u.surname AS therapist_name, t.specialization, COUNT(a.appointment_id) AS total_appointments, COUNT(a.appointment_id) FILTER (WHERE a.status = 'completed') AS completed_appointments, ROUND( 100.0 * COUNT(a.appointment_id) FILTER (WHERE a.status = 'completed') / NULLIF(COUNT(a.appointment_id), 0), 2 ) AS completion_rate_percent FROM therapists t JOIN users u ON u.user_id = t.therapist_id LEFT JOIN appointments a ON a.therapist_id = t.therapist_id GROUP BY t.therapist_id, u.name, u.surname, t.specialization ORDER BY total_appointments DESC, completion_rate_percent DESC; -- 9. Average feedback rating by therapist SELECT t.therapist_id, u.name || ' ' || u.surname AS therapist_name, t.specialization, COUNT(f.feedback_id) AS feedback_count, ROUND(AVG(f.rating), 2) AS average_rating FROM therapists t JOIN users u ON u.user_id = t.therapist_id LEFT JOIN feedbacks f ON f.therapist_id = t.therapist_id GROUP BY t.therapist_id, u.name, u.surname, t.specialization HAVING COUNT(f.feedback_id) > 0 ORDER BY average_rating DESC, feedback_count DESC; -- 10. Recent low-rating feedback that may need follow-up SELECT f.feedback_id, f.entry_date, f.rating, pu.name || ' ' || pu.surname AS patient_name, tu.name || ' ' || tu.surname AS therapist_name, f.comment FROM feedbacks f JOIN users pu ON pu.user_id = f.patient_id JOIN users tu ON tu.user_id = f.therapist_id WHERE f.rating <= 2 ORDER BY f.entry_date DESC, f.rating; -- 11. Patients with high average stress or anxiety in daily records SELECT p.patient_id, u.name || ' ' || u.surname AS patient_name, ROUND(AVG(dr.stress_level), 2) AS avg_stress, ROUND(AVG(dr.anxiety_level), 2) AS avg_anxiety, ROUND(AVG(dr.mood_level), 2) AS avg_mood, COUNT(dr.record_id) AS record_count FROM daily_records dr JOIN patients p ON p.patient_id = dr.patient_id JOIN users u ON u.user_id = p.patient_id GROUP BY p.patient_id, u.name, u.surname HAVING AVG(dr.stress_level) >= 7 OR AVG(dr.anxiety_level) >= 7 ORDER BY GREATEST(AVG(dr.stress_level), AVG(dr.anxiety_level)) DESC; -- 12. Daily mood trend for one patient WITH params AS ( SELECT 1 AS selected_patient_id ) SELECT dr.entry_date, dr.energy_level, dr.anxiety_level, dr.stress_level, dr.mood_level FROM daily_records dr JOIN params p ON p.selected_patient_id = dr.patient_id ORDER BY dr.entry_date; -- 13. Active prescriptions with patient and therapist names SELECT pr.prescription_id, pu.name || ' ' || pu.surname AS patient_name, tu.name || ' ' || tu.surname AS therapist_name, pr.medication, pr.dosage, pr.frequency, pr.start_date, pr.end_date, pr.status FROM prescriptions pr JOIN users pu ON pu.user_id = pr.patient_id JOIN users tu ON tu.user_id = pr.therapist_id WHERE pr.status = 'Active' ORDER BY pr.end_date NULLS LAST, patient_name; -- 14. Prescriptions ending within the next 30 days SELECT pr.prescription_id, pu.name || ' ' || pu.surname AS patient_name, pr.medication, pr.end_date, pr.status FROM prescriptions pr JOIN users pu ON pu.user_id = pr.patient_id WHERE pr.end_date BETWEEN CURRENT_DATE AND CURRENT_DATE + INTERVAL '30 days' ORDER BY pr.end_date; -- 15. Support group membership counts SELECT sg.group_id, sg.group_name, sg.topic, COUNT(gm.membership_id) AS member_count FROM support_groups sg LEFT JOIN group_membership gm ON gm.group_id = sg.group_id GROUP BY sg.group_id, sg.group_name, sg.topic ORDER BY member_count DESC, sg.group_name; -- 16. Support groups joined by each patient SELECT p.patient_id, u.name || ' ' || u.surname AS patient_name, sg.group_name, sg.topic, gm.joined_date FROM group_membership gm JOIN patients p ON p.patient_id = gm.patient_id JOIN users u ON u.user_id = p.patient_id JOIN support_groups sg ON sg.group_id = gm.group_id ORDER BY patient_name, gm.joined_date DESC; -- 17. Resource catalog with reference counts SELECT r.resource_id, r.title, r.description, r.upload_date, COUNT(rr.reference_id) AS reference_count FROM resources r LEFT JOIN resource_references rr ON rr.resource_id = r.resource_id GROUP BY r.resource_id, r.title, r.description, r.upload_date ORDER BY r.upload_date DESC; -- 18. Therapist weekly schedules SELECT t.therapist_id, u.name || ' ' || u.surname AS therapist_name, ts.weekday, ts.shift_type FROM therapists_schedules ts JOIN therapists t ON t.therapist_id = ts.therapist_id JOIN users u ON u.user_id = t.therapist_id ORDER BY therapist_name, ts.weekday, ts.shift_type; -- 19. Therapist schedule overrides SELECT tso.override_id, tso.date, u.name || ' ' || u.surname AS therapist_name, tso.override_type, tso.is_available, COUNT(tsofs.freeslot_id) AS override_free_slots FROM therapists_schedules_overrides tso JOIN users u ON u.user_id = tso.therapist_id LEFT JOIN therapists_schedules_overrides_free_slots tsofs ON tsofs.override_id = tso.override_id GROUP BY tso.override_id, tso.date, u.name, u.surname, tso.override_type, tso.is_available ORDER BY tso.date DESC, therapist_name; -- 20. Available therapist slots for a specific date WITH params AS ( SELECT DATE '2026-05-01' AS target_date ), target_day AS ( SELECT p.target_date, trim(to_char(p.target_date, 'Day')) AS weekday_name FROM params p ), scheduled_slots AS ( SELECT t.therapist_id, u.name || ' ' || u.surname AS therapist_name, td.target_date, sl.slot_id, sl.start_time, sl.end_time FROM target_day td JOIN therapists_schedules sch ON lower(sch.weekday) = lower(td.weekday_name) JOIN therapists t ON t.therapist_id = sch.therapist_id AND t.is_active = TRUE JOIN users u ON u.user_id = t.therapist_id JOIN time_slots sl ON sl.shift_type = sch.shift_type ) SELECT ss.therapist_id, ss.therapist_name, ss.target_date, ss.slot_id, ss.start_time, ss.end_time FROM scheduled_slots ss LEFT JOIN therapists_schedules_overrides tso ON tso.therapist_id = ss.therapist_id AND tso.date = ss.target_date LEFT JOIN therapists_schedules_overrides_free_slots tsofs ON tsofs.override_id = tso.override_id AND tsofs.slot_id = ss.slot_id LEFT JOIN appointments a ON a.therapist_id = ss.therapist_id AND a.slot_id = ss.slot_id AND a.scheduled_at::date = ss.target_date WHERE COALESCE(tso.is_available, TRUE) = TRUE AND ( tso.override_id IS NULL OR tso.is_available = FALSE OR NOT EXISTS ( SELECT 1 FROM therapists_schedules_overrides_free_slots fs WHERE fs.override_id = tso.override_id ) OR tsofs.freeslot_id IS NOT NULL ) AND a.appointment_id IS NULL ORDER BY ss.therapist_name, ss.start_time; -- 21. New patients by month SELECT date_trunc('month', registration_date)::date AS registration_month, COUNT(*) AS new_patient_count FROM patients GROUP BY date_trunc('month', registration_date) ORDER BY registration_month; -- 22. Diagnosis distribution SELECT diagnosis, COUNT(*) AS patient_count, ROUND(100.0 * COUNT(*) / SUM(COUNT(*)) OVER (), 2) AS percentage_of_patients FROM patients GROUP BY diagnosis ORDER BY patient_count DESC, diagnosis; -- 23. Report table compared with appointment facts WITH appointment_months AS ( SELECT trim(to_char(scheduled_at, 'Month')) AS month_name, COUNT(*) AS calculated_total_appointments FROM appointments GROUP BY trim(to_char(scheduled_at, 'Month')) ), feedback_months AS ( SELECT trim(to_char(entry_date, 'Month')) AS month_name, ROUND(AVG(rating), 2) AS calculated_avg_rating FROM feedbacks GROUP BY trim(to_char(entry_date, 'Month')) ) SELECT r.report_id, r.month, r.written_date, r.total_appointments AS reported_total_appointments, COALESCE(am.calculated_total_appointments, 0) AS calculated_total_appointments, r.avg_rating AS reported_avg_rating, fm.calculated_avg_rating, r.new_patients AS reported_new_patients FROM report r LEFT JOIN appointment_months am ON am.month_name ILIKE r.month || '%' LEFT JOIN feedback_months fm ON fm.month_name ILIKE r.month || '%' ORDER BY r.written_date DESC;
